6. 保存到 Supabase（可选）

不填也能用（仅存浏览器本地）。填入后可将对局存到云端、历史可查。Key 只需 anon 公钥。
首次使用：在 Supabase 控制台建表（展开） 
create table if not exists sessions (
  id bigint generated by default as identity primary key,
  name text,
  payload jsonb,
  created_at timestamptz default now()
);
alter table sessions enable row level security;
create policy "anon read" on sessions for select to anon using (true);
create policy "anon insert" on sessions for insert to anon with check (true);
